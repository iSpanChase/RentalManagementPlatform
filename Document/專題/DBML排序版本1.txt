////////////////////////////////////////////////////
// Address
////////////////////////////////////////////////////
Table ADDRESS {
  address_id int [pk, not null, note: '地址 ID']
  district_id int [ref: > DISTRICT.district_id, note: '行政區 ID (FK)']
    Latitude decimal(9,6) [not null, note: "CHECK (Latitude BETWEEN -90 AND 90)"]
  Longitude decimal(9,6) [not null, note: "CHECK (Longitude BETWEEN -180 AND 180)"]
  street string [note: '路名與門牌，例如：復興南路一段 390 號 10 樓']
  created_at datetime [note: '建立紀錄時間']
  updated_at datetime [note: '最後更新時間']}

////////////////////////////////////////////////////
// Board Proposal
////////////////////////////////////////////////////
Table BOARD_PROPOSAL {
  proposal_id int [pk, not null, note: '提案 ID']
  board_id int [ref: > PROPERTY_MANAGEMENT_BOARD.board_id, note: '所屬看板 ID (FK)']
  created_at datetime [ref: > ROOM_LIST.price_per_night , note: '建立時間']
  proposer_id int [ref: > USER.user_id, note: '提案人 ID (FK)']
  message string [note: '提案內容，例如：調整垃圾車時段、門禁卡申請']
  proposed_price decimal [note: '建議金額，例：清潔加價 300 元']}

////////////////////////////////////////////////////
// Booking
////////////////////////////////////////////////////
Table BOOKING {
  booking_id int [pk, not null, note: '訂單 ID']
  coupon_id int [ref: > COUPON.coupon_id, note: '優惠券 ID (FK)']
  guest_id int [ref: > USER.user_id, note: '旅客 ID (FK, role=guest)']
  room_id int [ref: > ROOM_LIST.room_id, note: '房源 ID (FK)']
  order_number string
  check_in datetime [note: '入住時間，例如：2025-02-10 15:00']
  check_out datetime [note: '退房時間，例如：2025-02-12 11:00']
  total_price decimal [note: '實付總額，含清潔費、服務費、折扣']
  commission_rate_snapshot decimal
  points_earned int
  points_redeemed int
  status string [note: '訂單狀態：pending / confirmed / cancelled']
  created_at datetime [note: '建立時間']}

Table USER {
  user_id int [pk, not null, note: '使用者 ID']

  auto_subscribe bool [note: '自動訂閱，true/false']
  password_hash string [note: '雜湊密碼，禁止明碼存放']

  role string [note: '角色：guest / host / admin']
  created_at datetime [note: '建立時間，建議存 UTC（台灣時區 +8）']
  updated_at datetime [note: '最後更新時間']}

Table POSTS {
  posts_id bigint [pk, increment, ref: > USER.user_id]
  user_id bigint [not null]// FK → users.id
  category_id int [not null, ref: > CATEGORIES.categories_id] // FK → categories.id
  region_id int [not null, ref: > DISTRICT.district_id] // FK → regions.id
  title varchar(120) [not null]
  contact_name varchar(100) [not null]
  content text [not null]
  address varchar(255) // 選填
  contact_phone varchar(30)
  contact_email varchar(255)
  contact_note varchar(255)
  publish_at datetime
  expire_at datetime
  views int [default: 0]
  is_promoted boolean [default: false]
  status post_status [not null, default: 'draft']
  created_at datetime [not null]
  updated_at datetime [not null]
  deleted_at datetime

  Note: '公告主表（刊登/審核/公開/下架的核心）'
  Indexes {
    (status, region_id, created_at)
    (category_id, status)
    // 提醒：可在實作時於 MySQL 加 FULLTEXT(title, content)
  }
}

  to_status varchar(20) [not null]

  Note: '公告狀態變更歷程（審核/下架留痕）'
  Indexes {
    (post_id, created_at)
  }
}

Table ANOMALY_RULE {
  rule_id int [pk, not null, note: '規則 ID']
  rule_name varchar(100) [note: '規則名稱，例如：租金高於區域平均 50%']
  target_type varchar(50) [note: '檢測目標類型，例如：Room / Host / Booking']
  condition_expression text [note: '檢測條件，以 JSON 定義規則邏輯']
